Python syntax

Class and constructor

the shape of every class in this course
class BankAccount:                       # PascalCase noun
    interest_rate = 0.005                # class attribute: one copy, shared

    def __init__(self, number, owner):   # self is always first
        self.__number = number           # private: name-mangled
        self.__owner = owner             # stores the INSTANCE, not a copy
        self.__balance = 0.0
        self.__transactions = []         # composition: born here, dies here

    @property                            # read-only
    def number(self):
        return self.__number

    @property
    def balance(self):
        return self.__balance

    def deposit(self, amount: float) -> bool:    # verb, snake_case, type hints
        if not isinstance(amount, (int, float)) or amount <= 0:
            raise ValueError('amount must be a positive number')
        self.__balance += amount
        self.__transactions.append(Transaction('D', amount, self.__balance))
        return True

    @staticmethod                        # no self: a helper that needs no state
    def baht_to_usd(baht: float) -> float:
        return baht / 36.0

    @classmethod                         # an alternative constructor
    def open_with_deposit(cls, number, owner, amount):
        acct = cls(number, owner)
        acct.deposit(amount)
        return acct

    def __str__(self):                   # what print() shows
        return f'{self.__number} ({self.__owner.name}): {self.__balance:.2f}'

Inheritance

single and multiple
class SavingsAccount(BankAccount):
    def __init__(self, number, owner, rate=0.005):
        super().__init__(number, owner)   # ALWAYS call up
        self.__rate = rate

    def calculate_interest(self):         # overriding
        return self.balance * self.__rate


# multiple inheritance: every class takes **kwargs and forwards it once
class Contact:
    def __init__(self, name='', email='', **kwargs):
        super().__init__(**kwargs)
        self.name, self.email = name, email

class AddressHolder:
    def __init__(self, street='', city='', **kwargs):
        super().__init__(**kwargs)
        self.street, self.city = street, city

class Friend(Contact, AddressHolder):
    def __init__(self, phone='', **kwargs):
        super().__init__(**kwargs)
        self.phone = phone

print(Friend.__mro__)    # when in doubt, read the MRO

Abstract base class

forcing a promise
from abc import ABC, abstractmethod

class PaymentProcessor(ABC):
    @abstractmethod
    def pay(self, order): ...        # subclasses MUST implement this

class CreditProcessor(PaymentProcessor):
    def pay(self, order):
        order.status = 'paid'

PaymentProcessor()   # TypeError — abstract classes are not things

Exceptions

raise, never print
class InsufficientFunds(Exception):
    def __init__(self, balance, requested):
        self.balance = balance
        self.requested = requested
        super().__init__(f'balance {balance} < requested {requested}')

try:
    account.withdraw(5000)
except InsufficientFunds as e:
    screen.show(f'Short by {e.requested - e.balance}')

Design review checklist

Run your mini-project past this before you present it.

  • Any attribute without __?
  • Any getter or setter nobody calls?
  • Any print() or input() inside a class?
  • Any fact stored in two places?
  • Any object with no container holding it?
  • Any if on a type string where a subclass would do?
  • Any class you cannot describe in one sentence?
  • Any inheritance whose is-a sentence sounds wrong out loud?
  • Any method over about 15 lines, or with more than about three parameters?
  • Any class that builds a collaborator it could be handed?
